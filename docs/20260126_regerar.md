### 4.0 ETAPA 0: Merge dos Erros de Instrumentação ✅ CONCLUÍDA

**Objetivo:** Consolidar os 37 arquivos `instrument_errors.json` em um único arquivo.

**Input:** `results/batch-*/instrument_errors.json` (37 arquivos)
**Output:** `results_final/exp02_generic_instrument_errors.json`

**Script simples (concatenação de JSONs):**
```python
#!/usr/bin/env python3
"""Merge instrument errors from all batches."""
import json
from pathlib import Path

def merge_instrument_errors():
    merged = {}
    for json_file in Path("../results").glob("*/instrument_errors.json"):
        with open(json_file) as f:
            data = json.load(f)
            merged.update(data)

    with open("../results_final/exp02_generic_instrument_errors.json", "w") as f:
        json.dump(merged, f, indent=2)

    print(f"Merged {len(merged)} APKs with instrument errors")

if __name__ == "__main__":
    merge_instrument_errors()
```

**Referência de formato:** Similar a `ase-journal/dataset/results/instrument/exp01_generic_instrument_errors.json`

### 4.2 ETAPA 2: Criação do main.py

**Estrutura proposta:**

```python
#!/usr/bin/env python3
"""
Main script for regenerating rvsec-02 experiment results.
"""

import os
import time
from pathlib import Path

import config
from domain.models import ExperimentType, SpecType
from generators.coverage import CoverageManager, MopErrorManager
from parsers.all_methods import AllMethods
from parsers.exp02_parser import Exp02Parser
from utils.file_utils import LogcatFileManager


class ResultsGenerator:
    def __init__(self):
        self.logcat_dir = config.LOGCAT_EXP02_GENERIC_DIR
        self.all_methods_dir = config.ALL_METHODS_GENERIC

    def generate_all_reports(self):
        print("Generating reports for rvsec-02 Phase 3...")
        start_time = time.time()

        logcat_manager = LogcatFileManager(self.logcat_dir)
        apks = logcat_manager.get_all_apks()

        logcat_data = []
        coverage_data = []
        mop_errors_data = []

        error_manager = MopErrorManager()

        for i, apk in enumerate(apks):
            print(f"Processing APK ({i+1}/{len(apks)}): {apk}")

            # Load .methods file
            methods_path = Path(os.path.join(self.all_methods_dir, f"{apk}{config.EXTENSION_METHODS}"))
            if not methods_path.exists():
                print(f"  WARNING: .methods not found for {apk}, skipping")
                continue

            coverage_manager = CoverageManager(AllMethods(methods_path))

            # Process each logcat for this APK
            logcat_files = logcat_manager.get_by_apk(apk)
            for logcat_file in logcat_files:
                parser = Exp02Parser(logcat_file)
                coverage, mop_errors, errors, crashes = parser.parse_all_lines(verbose=False)

                cov_report = coverage_manager.process_cov_file(coverage, verbose=False)
                mop_report = error_manager.process_file(mop_errors, verbose=False)

                # Collect data
                logcat_info = {
                    "apk": apk,
                    "rep": logcat_file.repetition,
                    "timeout": logcat_file.timeout,
                    "tool": logcat_file.tool,
                    "cov_items": len(coverage),
                    "cov_unique": len(coverage_manager.coverage_data),
                    "mop_errors": mop_report["unique_errors"],
                    "crashes": crashes,
                    "mismatch": len(cov_report["mismatch"]),
                }
                logcat_data.append(logcat_info)
                coverage_data.extend(cov_report["coverage_data"])
                mop_errors_data.extend(mop_report["errors_data"])

        # Save CSVs
        self._save_csv("coverage.csv", coverage_data)
        self._save_csv("errors.csv", mop_errors_data)
        self._save_csv("logcat.csv", logcat_data)

        # Generate summary
        summary = self._generate_summary(logcat_data, coverage_data, mop_errors_data)
        self._save_csv("summary.csv", summary)

        end_time = time.time()
        print(f"Total time: {end_time - start_time:.2f} seconds")

    def _generate_summary(self, logcat_data, coverage_data, mop_errors_data):
        import pandas as pd

        task_keys = ['apk', 'rep', 'timeout', 'tool']
        cov_keys = ['cov_act', 'cov_class', 'cov_method', 'cov_reachable',
                    'cov_reaches_mop', 'cov_directly_reaches_mop']

        df_logcat = pd.DataFrame(logcat_data)[task_keys].drop_duplicates()

        if coverage_data:
            df_cov = pd.DataFrame(coverage_data)
            cov_final = df_cov.groupby(task_keys).last()[cov_keys].reset_index()
        else:
            cov_final = pd.DataFrame(columns=task_keys + cov_keys)

        if mop_errors_data:
            df_err = pd.DataFrame(mop_errors_data)
            err_count = df_err.groupby(task_keys).size().reset_index(name='errors')
        else:
            err_count = pd.DataFrame(columns=task_keys + ['errors'])

        summary = (df_logcat
                  .merge(cov_final, on=task_keys, how='left')
                  .merge(err_count, on=task_keys, how='left'))

        for col in cov_keys:
            summary[col] = summary[col].fillna(0)
        summary['errors'] = summary['errors'].fillna(0).astype(int)

        return summary.to_dict('records')

    def _save_csv(self, filename, data):
        import pandas as pd
        filepath = os.path.join(config.OUTPUT_DIR, filename)
        print(f"Saving {filename} ({len(data)} rows) to {filepath}")
        pd.DataFrame(data).to_csv(filepath, index=False)


if __name__ == "__main__":
    ResultsGenerator().generate_all_reports()
```

## 6. Problemas Conhecidos

### 6.1 Package Mismatch

Problema detectado em ~17% dos APKs onde o pacote detectado não corresponde ao pacote real do APK.

**Sintoma:** Arquivo .methods com ~82 bytes (apenas cabeçalho CSV)

**Solução:** Documentar e excluir da análise final.

### 6.2 Inner Class Normalization

Diferença entre notação runtime (AspectJ) e estática (Soot):
- Runtime: `TabLayout.Tab`
- Estático: `TabLayout$Tab`

**Solução:** `SignatureNormalizer` já implementado em `processors/signature_normalizer.py`

### 6.3 Timestamps Negativos

Em testes longos que passam da meia-noite, timestamps podem ficar negativos.

**Solução:** Já tratado no `Exp02Parser._calculate_time_since_start()`
